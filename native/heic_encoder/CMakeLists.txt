cmake_minimum_required(VERSION 3.25)
project(LiveViewerHeicEncoder LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_path(HEIF_INCLUDE_DIR libheif/heif.h REQUIRED)
find_library(HEIF_LIBRARY NAMES heif libheif REQUIRED)
add_executable(LiveViewer.HeicEncoder main.cpp)
target_include_directories(LiveViewer.HeicEncoder PRIVATE "${HEIF_INCLUDE_DIR}")
target_link_libraries(LiveViewer.HeicEncoder PRIVATE "${HEIF_LIBRARY}")
if(APPLE)
  set_target_properties(LiveViewer.HeicEncoder PROPERTIES INSTALL_RPATH "@executable_path")
elseif(UNIX)
  set_target_properties(LiveViewer.HeicEncoder PROPERTIES INSTALL_RPATH "$ORIGIN")
endif()
install(TARGETS LiveViewer.HeicEncoder RUNTIME DESTINATION .)
if(MSVC)
  # Carry the matching Microsoft runtime beside the isolated helper. A clean
  # Windows installation must not need a separate redistributable installer.
  set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION bin)
  include(InstallRequiredSystemLibraries)
endif()
